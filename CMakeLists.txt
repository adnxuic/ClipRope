# Explicit targets as well as ALL_BUILD must enter through the serialized workflow.
function(cliprope_guard_directory directory)
    get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
    foreach(target IN LISTS targets)
        get_target_property(kind "${target}" TYPE)
        if(kind STREQUAL "EXECUTABLE" OR kind STREQUAL "STATIC_LIBRARY")
            add_dependencies("${target}" cliprope_build_session)
        endif()
    endforeach()
    get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
    foreach(child IN LISTS children)
        cliprope_guard_directory("${child}")
    endforeach()
endfunction()
cliprope_guard_directory("${PROJECT_SOURCE_DIR}")
